# Postgres Commands:

### Read Table
SELECT *
FROM (
    SELECT *
    FROM computed_metrics
    ORDER BY historical_data_id DESC
    LIMIT 1000
) AS subquery
ORDER BY historical_data_id ASC;



### Clear Data

DO
$$
DECLARE
    table_name text;
BEGIN
    -- Loop through all tables
    FOR table_name IN
        SELECT tablename
        FROM pg_tables
        WHERE schemaname = 'public' -- Change 'public' if using a different schema
    LOOP
        EXECUTE FORMAT('TRUNCATE TABLE %I RESTART IDENTITY CASCADE', table_name);
    END LOOP;
END;
$$;
